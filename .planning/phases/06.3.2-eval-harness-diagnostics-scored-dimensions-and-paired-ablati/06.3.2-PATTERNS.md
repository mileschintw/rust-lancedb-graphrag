            status="error",

        "bootstrap_seed": 42.0,

    return DimensionResult(
        name="graph_ablation_delta",
        status="ok",
        score=mu,
        detail=detail,
        n=len(diffs),

    records: list[RunRecord],
    usable_filter_fn,

            name="vector_yield",

            n=0,

        name="vector_yield",
        status="ok",

**Bootstrap and Wilson CI implementations (stdlib only, lines 865–916 of AI-SPEC):**

```python
import math
import random
from statistics import mean

BOOTSTRAP_B = 10_000
BOOTSTRAP_SEED = 42


def bootstrap_mean_ci(
    diffs: list[float],
    *,
    seed: int = BOOTSTRAP_SEED,
    b: int = BOOTSTRAP_B,
) -> tuple[float, float, float]:
    """Compute mean and 95% percentile bootstrap CI (D-38)."""
    n = len(diffs)
    if n == 0:
        raise ValueError("n_pairs=0")
    mu = mean(diffs)
    if n == 1:
        return mu, mu, mu  # degenerate case
    
    rng = random.Random(seed)
    means: list[float] = []
    for _ in range(b):
        acc = 0.0
        for _ in range(n):
            acc += diffs[rng.randrange(n)]
        means.append(acc / n)
    
    means.sort()
    return mu, means[int(0.025 * b)], means[min(b - 1, int(0.975 * b))]


def wilson_ci(k: int, n: int, z: float = 1.96) -> tuple[float, float, float]:
    """Compute rate and Wilson 95% CI (D-38)."""
    if n <= 0:
        raise ValueError("n=0")
    p = k / n
    z2 = z * z
    den = 1.0 + z2 / n
    center = (p + z2 / (2 * n)) / den
    half = z * math.sqrt((p * (1 - p) + z2 / (4 * n)) / n) / den
    return p, max(0.0, center - half), min(1.0, center + half)


def percentile(xs: list[float], q: float) -> float:
    """Linear interpolation percentile (D-38)."""
    ys = sorted(xs)
    n = len(ys)
    if n == 1:
        return ys[0]
    pos = q * (n - 1)
    lo = int(pos)
    hi = min(lo + 1, n - 1)
    frac = pos - lo
    return ys[lo] * (1.0 - frac) + ys[hi] * frac
```

**Pairing join pattern (D-37, D-40, add to score_run or separate module):**

```python
def form_pairs(
    records: list[RunRecord],
    gold_map: dict[str, GoldQuestion],
    usable_fn,
) -> list[tuple[RunRecord, RunRecord]]:
    """Inner join on (question_id, arm) where both arms usable, nulls excluded."""
    by_qid: dict[str, dict[str, RunRecord]] = {}
    
    for record in records:
        if not usable_fn(record):
            continue
        if record.question_id not in gold_map:
            continue
        
        gold = gold_map[record.question_id]
        if gold.is_null:
            continue  # D-40: exclude nulls from pairing join
        
        qid = record.question_id
        arm = record.graph_arm
        
        if qid not in by_qid:
            by_qid[qid] = {}
        by_qid[qid][arm] = record
    
    # Inner join: both arms required
    pairs = [
        (by_qid[qid]["graph-on"], by_qid[qid]["graph-off"])
        for qid in by_qid
        if "graph-on" in by_qid[qid] and "graph-off" in by_qid[qid]
    ]
    
    return sorted(pairs, key=lambda p: p[0].question_id)  # deterministic for bootstrap
```

### DimensionResult Validator Constraint (D-38, D-43)

**BLOCKER #2 (from RESEARCH.md §3):** Current validator (`dimensions.py:33-41`) forbids `detail` on `status in ("skipped", "error")`, but D-38/D-43 require `detail` floats for pairing metadata even when `n_pairs=0`.

**Validator code (lines 26–44):**

```python
@model_validator(mode="after")
def _validate_consistency(self) -> Self:
    if self.status == "ok":
        if self.score is None:
            raise ValueError("status 'ok' requires a score")
        if self.reason is not None:
            raise ValueError("status 'ok' cannot have a reason")
    elif self.status in ("skipped", "error"):
        if self.score is not None:
            raise ValueError(
                f"status {self.status!r} cannot carry a score, got {self.score}"
            )
        if self.detail:  # <- CONFLICT: pairing metadata needs detail
            raise ValueError(
                f"status {self.status!r} cannot carry detail, got {self.detail}"
            )
        if not self.reason or not self.reason.strip():
            raise ValueError(f"status {self.status!r} requires a non-blank reason")
    return self
```

**Planner decides (3 options):**

- **Option A:** Relax validator globally — allow `detail` on error/skipped. Simplest. Constraint was written to prevent mis-formatted skipped; error rows with diagnostic counts is a different case.
- **Option B:** Move counts outside dimension — create `DeltaMetadata` type or top-level `report.paired_delta_metadata` field.
- **Option C:** Use `status="ok"` with a detail flag (e.g., `floor_miss=1.0`), losing explicitness that the delta failed to compute.

**Recommendation:** **Option A** — `detail: dict[str, float] = Field(default_factory=dict)` is already flexible; allowing it on error/skipped doesn't break anything downstream that filters on status.
